# Дорожная карта — Подробный план реализации

**Старт:** 1× Orange Pi Zero 1GB (192.168.1.7, root@20051996)
**Цель:** Полноценный сервер CS: Economic Warfare с 4 OPi

---

## Анализ нестыковок проекта (ИСПРАВЛЕНО)

### Технические нестыковки

| # | Проблема | Решение |
|---|----------|---------|
| 1 | **fetch() не работает в Trident (IE)** — MOTD использует ES5, но `fetch` появился в IE11 частично | Использовать **XMLHttpRequest** вместо fetch во всех MOTD-скриптах |
| 2 | **AmxxEasyHttp не подтверждён для ARM** — плагин может не компилироваться | Использовать **HTTP-плагин AMXX** (http2.amxx) или нативный `socket` |
| 3 | **ReHLDS ARM-сборка** — нужен конкретный источник | Использовать **ReHLDS от ReGameDLL** (https://github.com/Arkhan/ReHLDS) или **cajuclc/cstrike-docker** |
| 4 | **Telegram Stars Payments** — это не API для ботов, а система оплаты Telegram Mini Apps | Реализовать через **Telegram Bot Payments API** (Stars = внутренняя валюта Telegram, принимается через Invoice) |
| 5 | **4 OPi при старте** — дорого и сложно | **Старт с 1 OPi**, расширение до 4 по мере роста онлайна |

### Логические нестыковки

| # | Проблема | Решение |
|---|----------|---------|
| 6 | **12v12 штурм** — при новом сервере не наберётся 24 игрока | Начать с **5v5 скирмишей**, масштабировать до 12v12 при 30+ онлайне |
| 7 | **Скины +5% к урону** = Pay-to-win | Оставить **только косметические бонусы** (подсветка, эффекты), убрать бонусы к урону |
| 8 | **Редкие города** — если их 3 и нельзя захватить, кто ими владеет изначально? | Определять владельцев **вручную** или по **первому захвату** при запуске |
| 9 | **Бизнесы каждые 2 часа** — при offline-сервере это не работает | Бизнесы начисляют доход **только при нахождении игрока на сервере** (AFK-защита) |
| 10 | **Дорожная карта: 8 недель / 10 этапов** — не совпадает | Исправлено: **12 недель / 12 этапов** сetailed ниже |
| 11 | **Домен cs-economic.duckdns.org** — бесплатный DDNS, ненадёжный | На старте OK, но планировать **переход на платный домен** к этапу 6 |
| 12 | **БД MySQL на OPi #4** — 1GB RAM мало для MySQL + Nginx + бот | На старте использовать **SQLite**, переходить на MySQL при расширении |

```pawn
// addons/amxmodx/scripting/business_system.sma
#include <amxmodx>
#include <amxmisc>

#define PLUGIN "Economic Warfare - Business"
#define VERSION "1.0"
#define AUTHOR "CS:EW Team"

// Типы бизнесов
enum _:BusinessType {
    BIZ_WEAPON_DEPOT,    // Оружейный склад
    BIZ_SKIN_FACTORY,    // Завод скинов
    BIZ_NARCO_LAB,       // Нарколаборатория
    BIZ_BITCOIN_FARM     // Ферма биткоинов
};

// Данные бизнесов игрока
new g_iBusinessLevel[33][BusinessType];
new Float:g_fLastIncome[33][BusinessType];

// Базовый доход бизнесов (каждые 2 часа)
new const Float:g_fBaseIncome[BusinessType] = {
    5.0,    // Оружейный склад
    3.0,    // Завод скинов
    8.0,    // Нарколаборатория
    10.0    // Ферма биткоинов
};

// Множитель за уровень
stock Float:getMultiplier(level) {
    return 1.0 + (level - 1) * 0.5; // Уровень 1 = x1.0, уровень 10 = x5.5
}

public plugin_init() {
    register_plugin(PLUGIN, VERSION, AUTHOR);
    register_clcmd("say /business", "cmdShowBusiness");
    register_clcmd("say /biz", "cmdShowBusiness");
    
    // Таймер проверки дохода каждые 5 минут
    set_task(300.0, "checkBusinessIncome", _, _, _, "b");
}

public client_putinserver(id) {
    // Инициализация бизнесов (пока все на уровне 0)
    for(new i = 0; i < BusinessType; i++) {
        g_iBusinessLevel[id][i] = 0;
        g_fLastIncome[id][i] = get_gametime();
    }
}

public checkBusinessIncome() {
    new Float:fCurrentTime = get_gametime();
    
    for(new id = 1; id <= 32; id++) {
        if(!is_user_connected(id)) continue;
        
        for(new biz = 0; biz < BusinessType; biz++) {
            if(g_iBusinessLevel[id][biz] <= 0) continue;
            
            // Проверяем прошло ли 2 часа (7200 секунд)
            if(fCurrentTime - g_fLastIncome[id][biz] >= 7200.0) {
                new Float:fIncome = g_fBaseIncome[biz] * getMultiplier(g_iBusinessLevel[id][biz]);
                new iIncome = floatround(fIncome);
                
                // Начисляем
                // g_iCoins[id] += iIncome; // Подключить Economic Core
                
                client_print_color(id, print_chat, 
                    "^4[CS:EW] ^1Ваш бизнес принёс ^3%d ^1Coins!", iIncome);
                
                g_fLastIncome[id][biz] = fCurrentTime;
            }
        }
    }
}
```

```javascript
// js/main.js - ES5 совместимый
(function() {
    // Получаем SteamID из URL
    var params = new URLSearchParams(window.location.search);
    var steamid = params.get('steamid');
    
    // Загрузка страницы
    function loadPage(page) {
        var xhr = new XMLHttpRequest();
        xhr.open('GET', 'pages/' + page + '.html', true);
        xhr.onreadystatechange = function() {
            if (xhr.readyState === 4 && xhr.status === 200) {
                document.getElementById('content').innerHTML = xhr.responseText;
                initPage(page);
            }
        };
        xhr.send();
    }
    
    // API запрос (XMLHttpRequest вместо fetch!)
    function apiRequest(endpoint, callback) {
        var xhr = new XMLHttpRequest();
        xhr.open('GET', '/api/' + endpoint + '?steamid=' + steamid, true);
        xhr.onreadystatechange = function() {
            if (xhr.readyState === 4 && xhr.status === 200) {
                var data = JSON.parse(xhr.responseText);
                callback(data);
            }
        };
        xhr.send();
    }
    
    // Инициализация табов
    var tabs = document.querySelectorAll('.tab');
    for (var i = 0; i < tabs.length; i++) {
        tabs[i].addEventListener('click', function() {
            var page = this.getAttribute('data-page');
            loadPage(page);
            
            // Обновление активного таба
            var allTabs = document.querySelectorAll('.tab');
            for (var j = 0; j < allTabs.length; j++) {
                allTabs[j].classList.remove('active');
            }
            this.classList.add('active');
        });
    }
    
    // Загружаем главную страницу
    loadPage('home');
})();
```

```javascript
// js/map.js
var WarMap = {
    canvas: null,
    ctx: null,
    
    init: function() {
        this.canvas = document.getElementById('warCanvas');
        if (!this.canvas) return;
        this.ctx = this.canvas.getContext('2d');
        this.draw();
        var self = this;
        setInterval(function() { self.draw(); }, 5000);
    },
    
    draw: function() {
        var self = this;
        var xhr = new XMLHttpRequest();
        xhr.open('GET', '/api/war?steamid=' + steamid, true);
        xhr.onreadystatechange = function() {
            if (xhr.readyState === 4 && xhr.status === 200) {
                var data = JSON.parse(xhr.responseText);
                self.render(data);
            }
        };
        xhr.send();
    },
    
    render: function(data) {
        this.ctx.clearRect(0, 0, 800, 600);
        
        // Линии снабжения
        for (var i = 0; i < data.lines.length; i++) {
            var l = data.lines[i];
            this.ctx.beginPath();
            this.ctx.moveTo(l.x1, l.y1);
            this.ctx.lineTo(l.x2, l.y2);
            this.ctx.strokeStyle = '#666';
            this.ctx.lineWidth = 2;
            this.ctx.stroke();
        }
        
        // Города
```

### 4.1. Python Flask API

```python
# /opt/cs-economic/api/app.py
from flask import Flask, request, jsonify
import sqlite3
import json
from datetime import datetime

app = Flask(__name__)

DB_PATH = '/opt/cs-economic/data/economic_warfare.db'

def get_db():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn

@app.route('/api/player/<steamid>')
def get_player(steamid):
    db = get_db()
    player = db.execute(
        'SELECT * FROM players WHERE steamid = ?', (steamid,)
    ).fetchone()
    
    if player:
        return jsonify(dict(player))
    return jsonify({'error': 'Player not found'}), 404

@app.route('/api/war')
def get_war_map():
    db = get_db()
    cities = db.execute('SELECT * FROM cities').fetchall()
    lines = db.execute('SELECT * FROM supply_lines').fetchall()
    ats = db.execute('SELECT * FROM assault_teams WHERE status = "moving"').fetchall()
    
    return jsonify({
        'cities': [dict(c) for c in cities],
        'lines': [dict(l) for l in lines],
        'ATs': [dict(a) for a in ats]
    })

@app.route('/api/business/<steamid>')
def get_business(steamid):
    db = get_db()
    businesses = db.execute(
        'SELECT * FROM businesses WHERE steamid = ?', (steamid,)
    ).fetchall()
    return jsonify([dict(b) for b in businesses])

@app.route('/api/battle/result', methods=['POST'])
def battle_result():
    data = request.form
    winner_id = data.get('winner')
    loser_id = data.get('loser')
    points = int(data.get('points', 0))
    
    db = get_db()
    
    # Начисляем победителю
    db.execute(
        'UPDATE players SET coins = coins + ? WHERE steamid = ?',
        (points, winner_id)
    )
    
    # Снимаем с проигравшего
    db.execute(
        'UPDATE players SET coins = MAX(0, coins - ?) WHERE steamid = ?',
        (points // 2, loser_id)
    )
    
    db.commit()
    return jsonify({'status': 'ok'})

@app.route('/api/exchange/rate')
def get_exchange_rate():
    # Курс Coins/Stars (динамический)
    db = get_db()
    rate = db.execute('SELECT rate FROM exchange_rate ORDER BY id DESC LIMIT 1').fetchone()
    return jsonify({'rate': rate['rate'] if rate else 100})

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)
```

```python
    text = "🏢 Ваши бизнесы:\n\n"

        text += f"• {biz['name']} (Ур. {biz['level']})\n"

        text += f"• {city['name']}: {owner}\n"

    keyboard.add(
        InlineKeyboardButton("💰 Баланс", callback_data='balance'),

        InlineKeyboardButton("🏢 Бизнес", callback_data='business'),
        InlineKeyboardButton("⚔️ Клан", callback_data='clan'),
    )
    return keyboard

if __name__ == '__main__':
    from aiogram import executor
    executor.start_polling(dp, skip_updates=True)
```

```pawn
    }
```
